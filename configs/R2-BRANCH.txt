! ============================================================
!  R2-BRANCH  (Cisco 2911)  - Branch office router
!  Roles: DHCP for branch LAN, OSPF neighbour of R1, SSH
! ============================================================
hostname R2-BRANCH
no ip domain-lookup
enable secret Cisco@123
banner motd #Authorized access only. Campus Network - R2-BRANCH#
service password-encryption
!
ip domain-name campus.local
username admin privilege 15 secret Admin@123
crypto key generate rsa general-keys modulus 1024
ip ssh version 2
!
ip dhcp excluded-address 192.168.50.1 192.168.50.10
ip dhcp pool BRANCH
 network 192.168.50.0 255.255.255.0
 default-router 192.168.50.1
 dns-server 192.168.40.10
 domain-name campus.local
!
interface GigabitEthernet0/0
 description BRANCH-LAN
 ip address 192.168.50.1 255.255.255.0
 no shutdown
!
interface GigabitEthernet0/1
 description WAN-TO-R1-HQ
 ip address 10.0.0.2 255.255.255.252
 no shutdown
!
router ospf 1
 router-id 2.2.2.2
 network 192.168.50.0 0.0.0.255 area 0
 network 10.0.0.0 0.0.0.3 area 0
 passive-interface GigabitEthernet0/0
!
line console 0
 password Console@123
 login
 logging synchronous
line vty 0 4
 login local
 transport input ssh
 exec-timeout 10 0
!
end
write memory
